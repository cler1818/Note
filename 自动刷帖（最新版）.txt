    async function withLikeSearchDeadline(task, deadline) {
        const remaining = Number(deadline) - Date.now();
        if (!Number.isFinite(remaining) || remaining <= 0) {
            throw new LikeSearchTimeoutError();
        }

        let timer = null;
        try {
            return await Promise.race([
                Promise.resolve().then(task),
                new Promise(function (_, reject) {
                    timer = setTimeout(function () {
                        reject(new LikeSearchTimeoutError());
                    }, remaining);
                })
            ]);
        } finally {
            if (timer) clearTimeout(timer);
        }

    function shuffle(items) {
        const result = items.slice();
        for (let i = result.length - 1; i > 0; i--) {
            const j = Math.floor(Math.random() * (i + 1));
            const temp = result[i];
            result[i] = result[j];
            result[j] = temp;
        }
        return result;
    }

    function enableDrag(element) {
        let dragging = false;
        let offsetX = 0;
        let offsetY = 0;

        element.addEventListener("mousedown", function (event) {
            if (event.target && (event.target.tagName === "BUTTON" || event.target.tagName === "INPUT")) return;
            dragging = true;
            offsetX = event.clientX - element.offsetLeft;
            offsetY = event.clientY - element.offsetTop;
        });

        document.addEventListener("mousemove", function (event) {
            if (!dragging) return;
            element.style.left = event.clientX - offsetX + "px";
            element.style.top = event.clientY - offsetY + "px";
            element.style.right = "auto";
            element.style.bottom = "auto";
        });

        document.addEventListener("mouseup", function () {
            dragging = false;
        });
    }

    function getTopicInfoFromUrl(url) {
        try {
            const parsed = new URL(url, location.origin);
            const match = parsed.pathname.match(/^\/t\/[^/]+\/(\d+)(?:\/(\d+))?/);
            if (!match) return null;
            return {
                topicId: String(match[1]),
                postNumber: match[2] ? parseInt(match[2], 10) : 1
            };
        } catch (_) {
            return null;
        }
    }

    function getCurrentTopicInfo() {
        return getTopicInfoFromUrl(location.href);
    }

        return /^\/latest(?:\/|$)/.test(location.pathname);

    function buildCandidateUrl(candidate) {
        const slug = String(candidate.slug || "topic").trim() || "topic";
        const base = location.origin + "/t/" + encodeURIComponent(slug) + "/" + candidate.topicId;
        return candidate.postNumber > 1 ? base + "/" + candidate.postNumber : base;
    }

    async function requestJson(url) {
        let response;
        try {
            response = await fetch(url, {
                credentials: "same-origin",
                cache: "no-store",
                headers: { "Accept": "application/json" }
            });
        } catch (error) {
            throw new ActivityApiError("请求失败: " + (error && error.message ? error.message : String(error)));
        }

        let data = null;
        try {
            data = await response.json();
        } catch (_) {
            data = null;
        }

        return { response: response, data: data };
    }

    async function ensureLoggedIn() {
        let result;
        try {
            result = await requestJson("/session/current.json");
        } catch (error) {
            loggedInUsername = "";
            if (error instanceof ActivityApiError || error instanceof ForumLoginError) {
                throw error;
            }
            throw new ForumLoginError(error.message);
        }

        if (!result.response.ok || !result.data || !result.data.current_user || !result.data.current_user.username) {
            loggedInUsername = "";
            throw new ForumLoginError("论坛没有登录");
        }

        loggedInUsername = String(result.data.current_user.username).trim();
        return result.data.current_user;
    }

    function parseGithubNames(text) {
        const seen = new Set();
        const names = [];

        String(text || "").split(/\r?\n/).forEach(function (line) {
            const name = line.replace(/^\uFEFF/, "").trim();
            if (!name || name.startsWith("#") || name.startsWith("//")) return;
            const key = name.toLowerCase();
            if (seen.has(key)) return;
            seen.add(key);
            names.push(name);
        });

        return names;
    }

    function getCachedWhitelistNames() {
        const cached = readJson(KEY.WHITELIST_CACHE, null);
        if (Array.isArray(cached)) return cached;
        if (cached && Array.isArray(cached.names)) return cached.names;
        return [];
    }

    async function loadWhitelistNames() {
        try {
            const response = await fetch(GITHUB_LIST_URL, { cache: "no-store" });
            if (!response.ok) throw new Error("HTTP " + response.status);
            const names = parseGithubNames(await response.text());
            if (!names.length) throw new Error("名单为空");

            writeJson(KEY.WHITELIST_CACHE, {
                names: names,
                updatedAt: Date.now()
            });
            return names;
        } catch (error) {
            if (!running) return [];
            const cached = getCachedWhitelistNames();
            if (cached.length) {
                setMessage("GitHub 名单读取失败，已使用本地缓存");
                return cached;
            }

            console.warn("[自动刷帖] GitHub 名单读取失败:", error);
            setMessage("GitHub 名单读取失败且没有缓存，将只使用普通随机点赞");
            return [];
        }
    }

    async function fetchUserActionFilter(username, filter) {
        const url = "/user_actions.json?username=" + encodeURIComponent(username) +
            "&filter=" + filter + "&limit=30&offset=0";
        const result = await requestJson(url);

        if (result.response.status === 404) return [];
        if (result.response.status === 401 || result.response.status === 403) {
            throw new ForumLoginError("论坛没有登录");
        }
        if (!result.response.ok) {
            throw new ActivityApiError("动态接口返回 HTTP " + result.response.status);
        }
        if (!result.data || !Array.isArray(result.data.user_actions)) {
            throw new ActivityApiError("动态接口返回格式异常");
        }

        return result.data.user_actions;
    }

    function normalizeActivityItem(item, username) {
        if (!item || item.deleted || item.hidden) return null;
        if (Number(item.action_type) !== 4 && Number(item.action_type) !== 5) return null;
        if (!item.topic_id) return null;
        if (!item.username || String(item.username).toLowerCase() !== String(username).toLowerCase()) return null;

        const parsedPostNumber = parseInt(item.post_number || "1", 10);
        const postNumber = Number.isFinite(parsedPostNumber) ? Math.max(1, parsedPostNumber) : 1;
        const candidate = {
            username: String(username),
            topicId: String(item.topic_id),
            postNumber: postNumber,
            postId: item.post_id ? String(item.post_id) : null,
            slug: item.slug || item.topic_slug || "topic",
            title: item.title || "",
            createdAt: item.created_at || ""
        };
        candidate.url = buildCandidateUrl(candidate);
        return candidate;
    }

    async function fetchRecentCandidatesForUser(username) {
        const results = await Promise.all([
            fetchUserActionFilter(username, 4),
            fetchUserActionFilter(username, 5)
        ]);

        const merged = results[0].concat(results[1])
            .map(function (item) {
                return normalizeActivityItem(item, username);
            })
            .filter(Boolean)
            .sort(function (a, b) {
                const bTime = Date.parse(b.createdAt || "") || 0;
                const aTime = Date.parse(a.createdAt || "") || 0;
                return bTime - aTime;
            });

        const uniquePosts = [];
        const seen = new Set();
        for (let i = 0; i < merged.length; i++) {
            const item = merged[i];
            const key = item.topicId + ":" + item.postNumber;
            if (seen.has(key)) continue;
            seen.add(key);
            uniquePosts.push(item);
            if (uniquePosts.length >= 30) break;
        }

        return uniquePosts;
    }

    async function refreshWhitelistNames(force) {
        if (whitelistRefreshPromise) return whitelistRefreshPromise;

        whitelistRefreshPromise = (async function () {
            if (!force && whitelistNames.length && whitelistRefreshAt &&
                Date.now() - whitelistRefreshAt < CONFIG.WHITELIST_REFRESH_INTERVAL) {
                return whitelistNames;
            }

            whitelistNames = await loadWhitelistNames();
            whitelistRefreshAt = Date.now();
            writeJson(KEY.WHITELIST_NAMES, whitelistNames);
            localStorage.setItem(KEY.WHITELIST_REFRESH, String(whitelistRefreshAt));
            return whitelistNames;
        })();

        try {
            return await whitelistRefreshPromise;
        } finally {
            whitelistRefreshPromise = null;
        }
    }

    function normalizeTriedUsers(triedUsers) {
        const result = [];
        const seen = new Set();
        (Array.isArray(triedUsers) ? triedUsers : []).forEach(function (username) {
            const name = String(username || "").trim();
            const key = name.toLowerCase();
            if (!name || seen.has(key)) return;
            seen.add(key);
            result.push(name);
        });
        return result;
    }

    function normalizeWhitelistProgress(value) {
        if (!value || !Array.isArray(value.users)) return null;

        const users = normalizeTriedUsers(value.users);
        if (!users.length) return null;

        const rawStates = Array.isArray(value.states) ? value.states : [];
        const stateByUsername = new Map();

        rawStates.forEach(function (rawState) {
            const username = String(rawState && rawState.username || "").trim();
            if (!username) return;

            const candidates = Array.isArray(rawState.candidates)
                ? rawState.candidates.filter(function (candidate) {
                    return candidate && candidate.topicId;
                })
                : [];
            const parsedCandidateIndex = parseInt(rawState.candidateIndex || "0", 10);
            const candidateIndex = Number.isFinite(parsedCandidateIndex)
                ? Math.max(0, Math.min(candidates.length, parsedCandidateIndex))
                : 0;

            stateByUsername.set(username.toLowerCase(), {
                username: username,
                candidates: candidates,
                candidateIndex: candidateIndex,
                loaded: rawState.loaded === true || candidates.length > 0,
                exhausted: rawState.exhausted === true
            });
        });

        const states = users.map(function (username) {
            const existing = stateByUsername.get(username.toLowerCase());
            if (existing) {
                existing.username = username;
                return existing;
            }

            return {
                username: username,
                candidates: [],
                candidateIndex: 0,
                loaded: false,
                exhausted: false
            };
        });

        const requestedLastUsername = String(value.lastUsername || "").trim();
        const lastUsername = states.some(function (state) {
            return requestedLastUsername &&
                state.username.toLowerCase() === requestedLastUsername.toLowerCase();
        }) ? requestedLastUsername : null;

        return {
            users: users,
            states: states,
            lastUsername: lastUsername
        };
    }

    function saveWhitelistProgress(value) {
        whitelistProgress = normalizeWhitelistProgress(value);
        if (whitelistProgress) {
            writeJson(KEY.WHITELIST_PROGRESS, whitelistProgress);
        } else {
            localStorage.removeItem(KEY.WHITELIST_PROGRESS);
        }
    }

    function beginWhitelistRound() {
        saveWhitelistProgress(null);
    }

    function filterWhitelistNamesForCurrentUser(names) {
        return normalizeTriedUsers(names).filter(function (username) {
            return !isCurrentLoggedInUser(username);
        });
    }

    function isCurrentLoggedInUser(username) {
        const selfKey = String(loggedInUsername || "").trim().toLowerCase();
        const candidateKey = String(username || "").trim().toLowerCase();
        return !!selfKey && !!candidateKey && selfKey === candidateKey;
    }

    // 只从白名单候选中排除当前登录账号；普通随机点赞兜底不受影响。
    function createLikeSearchDeadline() {
        return Date.now() + CONFIG.MAX_LIKE_SEARCH_DURATION;
    }

    function normalizeLikeSearchDeadline(value) {
        const deadline = Number(value);
        return Number.isFinite(deadline) && deadline > 0 ? deadline : createLikeSearchDeadline();
    }

    function isLikeSearchExpired(deadline) {
        return Number.isFinite(Number(deadline)) && Date.now() >= Number(deadline);
    }

    async function chooseRandomWhitelistCandidate(triedUsers, searchDeadline) {
        // triedUsers 参数仅保留给旧导航状态；当前版本会在每次候选失败后
        // 优先换到另一位随机用户，再从其最近 30 条中抽取一条。
        void triedUsers;
        const deadline = normalizeLikeSearchDeadline(searchDeadline);

        if (isLikeSearchExpired(deadline)) {
            return { candidate: null, triedUsers: [], timedOut: true };
        }

        let progress = normalizeWhitelistProgress(whitelistProgress);
        if (progress && loggedInUsername &&
            progress.users.some(function (username) {
                return isCurrentLoggedInUser(username);
            })) {
            // 兼容旧版本或切换登录账号后的残留进度，重新生成本轮白名单顺序。
            saveWhitelistProgress(null);
            progress = null;
        }

        if (!progress) {
            let names;
            try {
                names = await withLikeSearchDeadline(function () {
                    return refreshWhitelistNames(false);
                }, deadline);
            } catch (error) {
                if (error instanceof LikeSearchTimeoutError) {
                    return { candidate: null, triedUsers: [], timedOut: true };
                }
                throw error;
            }
            if (isLikeSearchExpired(deadline)) {
                return { candidate: null, triedUsers: [], timedOut: true };
            }

            progress = {
                users: shuffle(filterWhitelistNamesForCurrentUser(names)),
                states: [],
                lastUsername: lastWhitelistUser || null
            };
            saveWhitelistProgress(progress);
            progress = normalizeWhitelistProgress(whitelistProgress);
        }

        while (progress && progress.states.length) {
            if (!running) {
                return { candidate: null, triedUsers: [], timedOut: false };
            }
            if (isLikeSearchExpired(deadline)) {
                return {
                    candidate: null,
                    triedUsers: [],
                    timedOut: true
                };
            }

            let selectableStates = progress.states.filter(function (state) {
                return !state.exhausted && !isCurrentLoggedInUser(state.username);
            });

            if (!selectableStates.length) break;

            if (selectableStates.length > 1 && progress.lastUsername) {
                const otherUsers = selectableStates.filter(function (state) {
                    return state.username.toLowerCase() !== progress.lastUsername.toLowerCase();
                });
                if (otherUsers.length) selectableStates = otherUsers;
            }

            const state = selectableStates[randomBetween(0, selectableStates.length - 1)];
            const username = state.username;

            if (!state.loaded) {
                setMessage("本轮随机到白名单用户 " + username + "，正在读取最近 30 条内容");
                let recentItems;
                try {
                    recentItems = await withLikeSearchDeadline(function () {
                        return fetchRecentCandidatesForUser(username);
                    }, deadline);
                } catch (error) {
                    if (error instanceof LikeSearchTimeoutError) {
                        return {
                            candidate: null,
                            triedUsers: [username],
                            timedOut: true
                        };
                    }
                    throw error;
                }
                if (isLikeSearchExpired(deadline)) {
                    return {
                        candidate: null,
                        triedUsers: [username],
                        timedOut: true
                    };
                }

                state.candidates = shuffle(recentItems).slice(0, 30);
                state.candidateIndex = 0;
                state.loaded = true;
                state.exhausted = state.candidates.length === 0;
                saveWhitelistProgress(progress);
            }

            while (state.candidateIndex < state.candidates.length) {
                if (!running) {
                    return { candidate: null, triedUsers: [], timedOut: false };
                }
                if (isLikeSearchExpired(deadline)) {
                    return {
                        candidate: null,
                        triedUsers: [username],
                        timedOut: true
                    };
                }

                const candidate = state.candidates[state.candidateIndex];
                state.candidateIndex++;
                saveWhitelistProgress(progress);

                if (!candidate || !candidate.topicId ||
                    handledLikeTopics.has(String(candidate.topicId))) {
                    continue;
                }

                if (!candidate.url) candidate.url = buildCandidateUrl(candidate);
                progress.lastUsername = username;
                lastWhitelistUser = username;
                localStorage.setItem(KEY.LAST_WHITELIST_USER, lastWhitelistUser);
                saveWhitelistProgress(progress);
                setMessage(
                    "随机尝试白名单用户 " + username +
                    " 的一条内容（候选 " + state.candidateIndex + "/" + state.candidates.length + "）"
                );
                return {
                    candidate: candidate,
                    triedUsers: [username],
                    timedOut: false
                };
            }

            state.exhausted = true;
            saveWhitelistProgress(progress);
            setMessage(username + " 的最近 30 条没有剩余候选，正在随机下一位白名单用户");
        }

        return {
            candidate: null,
            triedUsers: progress ? progress.users.slice() : [],
            timedOut: false
        };
    }

    function markLikeTopicHandled(topicId) {
        const id = String(topicId);
        handledLikeTopics.add(id);
        writeJson(KEY.HANDLED_TOPICS, Array.from(handledLikeTopics));
    }

    function scheduleNextLike(baseTime) {
        const base = Number.isFinite(baseTime) ? baseTime : Date.now();
        if (likeCooldownUntil > Date.now()) {
            nextLikeAt = likeCooldownUntil;
        } else {
            nextLikeAt = base + randomBetween(
                CONFIG.MIN_LIKE_INTERVAL,
                CONFIG.MAX_LIKE_INTERVAL
            );
        }
        localStorage.setItem(KEY.NEXT_LIKE, String(nextLikeAt));
    }

    function ensureLikeSchedule() {
        if (!Number.isFinite(nextLikeAt) || nextLikeAt <= 0) {
            scheduleNextLike(Date.now());
        }
    }

    function isLikeDue() {
        const now = Date.now();
        return running && now >= nextLikeAt && now >= likeCooldownUntil;
    }

    function isLikeButtonLiked(button) {
        if (!button) return false;
        const stateLabel = [
            button.getAttribute("title") || "",
            button.getAttribute("aria-label") || "",
            button.getAttribute("data-tooltip") || ""
        ].join(" ").toLowerCase();

        return button.classList.contains("has-like") ||
            button.classList.contains("liked") ||
            button.classList.contains("is-active") ||
            button.getAttribute("aria-pressed") === "true" ||
            button.getAttribute("data-checked") === "true" ||
            stateLabel.includes("移除此赞") ||
            stateLabel.includes("取消点赞") ||
            stateLabel.includes("撤销点赞") ||
            stateLabel.includes("收回点赞") ||
            stateLabel.includes("移除您自己的回应") ||
            stateLabel.includes("无法再移除") ||
            stateLabel.includes("remove this like") ||
            stateLabel.includes("remove like") ||
            stateLabel.includes("remove your own reaction") ||
            stateLabel.includes("can no longer remove") ||
            stateLabel.includes("unlike");
    }

    function getAllLikeButtons(scope) {
        const root = scope || document;
        const found = new Set();

        Array.from(root.querySelectorAll("button.btn-toggle-reaction-like")).forEach(function (button) {
            found.add(button);
        });

        Array.from(root.querySelectorAll(".discourse-reactions-reaction-button button")).forEach(function (button) {
            const label = [
                button.getAttribute("data-reaction-name") || "",
                button.getAttribute("aria-label") || "",
                button.getAttribute("title") || ""
            ].join(" ").toLowerCase();

            if (button.classList.contains("btn-toggle-reaction-like") ||
                label.includes("like") ||
                label.includes("点赞") ||
                label.includes("赞")) {
                found.add(button);
            }
        });

        return Array.from(found);
    }

    function getAvailableLikeButtons(scope) {
        return getAllLikeButtons(scope).filter(function (button) {
            return button.isConnected &&
                !button.disabled &&
                button.getAttribute("aria-disabled") !== "true" &&
                !isLikeButtonLiked(button);
        });
    }

    function getLikeButtonKey(button) {
        if (!button) return "";
        const reactionName = button.getAttribute("data-reaction-name") || "";
        const reactionId = button.getAttribute("data-reaction-id") || "";
        if (reactionName || reactionId) {
            return (reactionName + "|" + reactionId).toLowerCase();
        }
        return button.classList.contains("btn-toggle-reaction-like")
            ? "btn-toggle-reaction-like"
            : "";
    }

    function hasConfirmedLike(scope, originalButton, originalKey) {
        if (originalButton && originalButton.isConnected) {
            return isLikeButtonLiked(originalButton);
        }

        const buttons = getAllLikeButtons(scope);
        if (originalKey) {
            const replacement = buttons.find(function (button) {
                return getLikeButtonKey(button) === originalKey;
            });
            if (replacement) return isLikeButtonLiked(replacement);
        }

        // 只有目标帖子只有一个点赞按钮时才允许无标识兜底，避免把同帖其他反应误判为成功。
        return buttons.length === 1 && isLikeButtonLiked(buttons[0]);
    }

    async function waitForConfirmedLike(scope, originalButton, originalKey) {
        const deadline = Date.now() + CONFIG.LIKE_CONFIRM_WAIT;
        let stableRounds = 0;
        while (Date.now() < deadline && running) {
            if (hasConfirmedLike(scope, originalButton, originalKey)) {
                stableRounds++;
                if (stableRounds >= 2) return true;
            } else {
                stableRounds = 0;
            }
            if (likeCooldownUntil > Date.now()) return false;
            await sleep(250);
        }
        return hasConfirmedLike(scope, originalButton, originalKey);
    }

    function findTargetPostContainer(candidate) {
        const posts = Array.from(document.querySelectorAll(".topic-post"));

        if (candidate.postId) {
            const byPostId = posts.find(function (post) {
                return post.getAttribute("data-post-id") === String(candidate.postId);
            });
            if (byPostId) return byPostId;
        }

        const byNumber = posts.find(function (post) {
            return post.getAttribute("data-post-number") === String(candidate.postNumber);
        });
        if (byNumber) return byNumber;

        if (candidate.postNumber === 1 && posts.length === 1) return posts[0];
        return null;
    }

    async function waitForTargetPost(candidate, searchDeadline) {
        const deadline = Math.min(
            Date.now() + CONFIG.TARGET_POST_WAIT,
            normalizeLikeSearchDeadline(searchDeadline)
        );
        let lastPost = null;

        while (Date.now() < deadline && running) {
            lastPost = findTargetPostContainer(candidate);
            if (lastPost && getAllLikeButtons(lastPost).length) {
                return lastPost;
            }
            await sleep(250);
        }

        return lastPost;
    }

    async function waitForGenericLikeButtons(searchDeadline) {
        const deadline = Math.min(
            Date.now() + CONFIG.LIKE_BUTTON_WAIT,
            normalizeLikeSearchDeadline(searchDeadline)
        );
        while (Date.now() < deadline && running) {
            const buttons = getAvailableLikeButtons(document);
            if (buttons.length) return buttons;
            await sleep(250);
        }
        return getAvailableLikeButtons(document);
    }

    async function performLikeButton(button, topicId, sourceLabel, searchDeadline) {
        const id = String(topicId);
        if (!running) return { status: "stopped" };
        if (handledLikeTopics.has(id)) return { status: "handled" };
        if (isLikeSearchExpired(searchDeadline)) return { status: "timeout" };

        const postContainer = button.closest(".topic-post") || document;
        const originalKey = getLikeButtonKey(button);

        if (isLikeButtonLiked(button)) {
            markLikeTopicHandled(id);
            return { status: "already" };
        }

        button.scrollIntoView({ behavior: "smooth", block: "center" });
        setMessage(sourceLabel + "，准备点赞...");
        await sleep(randomBetween(CONFIG.PRE_CLICK_MIN, CONFIG.PRE_CLICK_MAX));

        if (!running) return { status: "stopped" };
        if (isLikeSearchExpired(searchDeadline)) return { status: "timeout" };
        if (!button.isConnected || isLikeButtonLiked(button)) {
            markLikeTopicHandled(id);
            return { status: "already" };
        }

        markLikeTopicHandled(id);

        likeClickInProgress = true;
        try {
            try {
                button.click();
            } catch (error) {
                console.warn("[自动刷帖] 点赞按钮点击失败:", error);
                return { status: "failed" };
            }

            await sleep(randomBetween(CONFIG.AFTER_CLICK_MIN, CONFIG.AFTER_CLICK_MAX));

            const confirmed = await waitForConfirmedLike(postContainer, button, originalKey);

            if (confirmed) {
                likeCount++;
                localStorage.setItem(KEY.LIKE_COUNT, String(likeCount));
                updateUI();
                setMessage(sourceLabel + "点赞成功，本次已点赞 " + likeCount + " 次");
                return { status: "success" };
            }

            if (likeCooldownUntil > Date.now()) {
                return { status: "rate_limited" };
            }

            setMessage(sourceLabel + "点赞未确认，继续寻找下一个目标");
            return { status: "failed" };
        } finally {
            likeClickInProgress = false;
        }
    }

    async function attemptTargetCandidate(candidate, searchDeadline) {
        const topicId = String(candidate.topicId);
        if (handledLikeTopics.has(topicId)) return { status: "handled" };
        if (!isCandidateTopicPage(candidate)) return { status: "wrong_page" };
        if (isLikeSearchExpired(searchDeadline)) return { status: "timeout" };
        if (isCurrentLoggedInUser(candidate.username)) return { status: "self" };

        setMessage("正在定位名单用户 " + candidate.username + " 的帖子...");
        const post = await waitForTargetPost(candidate, searchDeadline);
        if (!running) return { status: "stopped" };
        if (isLikeSearchExpired(searchDeadline)) return { status: "timeout" };

        if (!post) {
            markLikeTopicHandled(topicId);
            setMessage("未找到 " + candidate.username + " 的目标楼层，继续找下一位");
            return { status: "unavailable" };
        }

        const allButtons = getAllLikeButtons(post);
        const available = getAvailableLikeButtons(post);

        if (!available.length) {
            markLikeTopicHandled(topicId);
            if (allButtons.some(isLikeButtonLiked)) {
                setMessage(candidate.username + " 的该主题已经点赞，继续找下一位");
                return { status: "already" };
            }
            setMessage(candidate.username + " 的该楼层无法点赞，继续找下一位");
            return { status: "unavailable" };
        }

        const button = available[Math.floor(Math.random() * available.length)];
        return performLikeButton(
            button,
            topicId,
            "名单用户 " + candidate.username + " ",
            searchDeadline
        );
    }

    async function attemptGenericLike(topicId, searchDeadline) {
        const id = String(topicId);
        if (handledLikeTopics.has(id)) return { status: "handled" };
        if (isLikeSearchExpired(searchDeadline)) return { status: "timeout" };

        const buttons = await waitForGenericLikeButtons(searchDeadline);
        if (!running) return { status: "stopped" };
        if (isLikeSearchExpired(searchDeadline)) return { status: "timeout" };

        if (!buttons.length) {
            markLikeTopicHandled(id);
            setMessage("本主题没有可点赞内容，继续刷帖");
            return { status: "unavailable" };
        }

        const button = buttons[Math.floor(Math.random() * buttons.length)];
        return performLikeButton(button, id, "普通随机", searchDeadline);
    }

    function isLikeRequestUrl(url) {
        const text = String(url || "");
        return text.includes("/discourse-reactions/") || text.includes("/toggle.json");
    }

    function isReadTimingRequestUrl(url) {
        return String(url || "").includes("/topics/timings");
    }

    function extractReadTimingData(body) {
        const postNumbers = new Set();
        let topicId = "";

        function consume(key, value) {
            const name = String(key || "");
            const timingMatch = name.match(/^timings\[(\d+)\]$/);
            if (timingMatch && Number(value) > 0) {
                postNumbers.add(parseInt(timingMatch[1], 10));
            }
            if (name === "topic_id" && value !== undefined && value !== null) {
                topicId = String(value);
            }
        }

        try {
            if (typeof body === "string") {
                const text = body.trim();
                if (text.charAt(0) === "{") {
                    const parsed = JSON.parse(text);
                    topicId = parsed && parsed.topic_id ? String(parsed.topic_id) : "";
                    if (parsed && parsed.timings && typeof parsed.timings === "object") {
                        Object.keys(parsed.timings).forEach(function (number) {
                            consume("timings[" + number + "]", parsed.timings[number]);
                        });
                    }
                } else if (text) {
                    new URLSearchParams(text).forEach(function (value, key) {
                        consume(key, value);
                    });
                }
            } else if (typeof URLSearchParams !== "undefined" && body instanceof URLSearchParams) {
                body.forEach(function (value, key) {
                    consume(key, value);
                });
            } else if (typeof FormData !== "undefined" && body instanceof FormData) {
                body.forEach(function (value, key) {
                    consume(key, value);
                });
            } else if (body && typeof body === "object") {
                topicId = body.topic_id ? String(body.topic_id) : "";
                if (body.timings && typeof body.timings === "object") {
                    Object.keys(body.timings).forEach(function (number) {
                        consume("timings[" + number + "]", body.timings[number]);
                    });
                }
            }
        } catch (error) {
            console.debug("[自动刷帖] 读取原生已读请求参数失败:", error);
        }

        return {
            topicId: topicId,
            postNumbers: Array.from(postNumbers).filter(function (number) {
                return Number.isFinite(number) && number > 1;
            })
        };
    }

    function recordConfirmedReadTiming(topicId, postNumbers) {
        // 只统计本脚本当前运行期间的论坛原生确认，避免停止后手动浏览污染本轮数据。
        if (!running) return;
        const id = String(topicId || getCurrentTopicId() || "");
        if (!id || !Array.isArray(postNumbers) || !postNumbers.length) return;

        let changed = 0;
        postNumbers.forEach(function (postNumber) {
            const number = parseInt(postNumber, 10);
            if (!Number.isFinite(number) || number <= 1) return;
            const key = id + ":" + number;
            if (confirmedReadReplies.has(key)) return;
            confirmedReadReplies.add(key);
            changed++;
        });

        if (!changed) return;
        readReplyCount += changed;
        localStorage.setItem(KEY.READ_REPLY_COUNT, String(readReplyCount));
        writeJson(KEY.READ_REPLY_KEYS, Array.from(confirmedReadReplies));
        updateUI();
    }

    function handleLikeRateLimit(data) {
        const seconds = Number(data && data.extras && data.extras.wait_seconds) ||
            CONFIG.DEFAULT_RATE_LIMIT_SECONDS;
        likeCooldownUntil = Date.now() + Math.max(1, seconds) * 1000;
        nextLikeAt = likeCooldownUntil;
        localStorage.setItem(KEY.LIKE_COOLDOWN, String(likeCooldownUntil));
        localStorage.setItem(KEY.NEXT_LIKE, String(nextLikeAt));

        const minutes = Math.max(1, Math.ceil(seconds / 60));
        setMessage("点赞已受站点限制，将暂停 " + minutes + " 分钟");
        console.warn("[自动刷帖] 点赞触发限制，暂停 " + minutes + " 分钟");
    }

    function installLikeLimitInterceptor() {
        if (window.__ldAutoLikeV3InterceptorInstalled) return;
        window.__ldAutoLikeV3InterceptorInstalled = true;

        if (typeof window.fetch === "function") {
            const originalFetch = window.fetch;
            window.fetch = async function () {
                const args = Array.from(arguments);
                const input = args[0];
                const init = args[1] || {};
                const url = typeof input === "string" ? input :
                    (input && (input.url || input.href)) || "";
                let timingBodyPromise = Promise.resolve(init.body);

                // Request 的正文可能在 fetch 发出后就被消费，必须提前 clone 保存。
                if (isReadTimingRequestUrl(url) && init.body === undefined &&
                    input && typeof input.clone === "function") {
                    try {
                        timingBodyPromise = input.clone().text().catch(function () {
                            return null;
                        });
                    } catch (_) {
                        timingBodyPromise = Promise.resolve(null);
                    }
                }

                const response = await originalFetch.apply(this, args);

                if (isLikeRequestUrl(url) && response.status === 429) {
                    response.clone().json()
                        .then(handleLikeRateLimit)
                        .catch(function () {
                            handleLikeRateLimit(null);
                        });
                }

                if (isReadTimingRequestUrl(url) && response.status >= 200 && response.status < 300) {
                    const body = await timingBodyPromise;
                    const timing = extractReadTimingData(body);
                    recordConfirmedReadTiming(timing.topicId, timing.postNumbers);
                }
                return response;
            };
        }

        const originalOpen = XMLHttpRequest.prototype.open;
        const originalSend = XMLHttpRequest.prototype.send;

        XMLHttpRequest.prototype.open = function (method, url) {
            this.__ldAutoLikeUrl = url;
            return originalOpen.apply(this, arguments);
        };

        XMLHttpRequest.prototype.send = function () {
            const requestBody = arguments[0];
            if (isLikeRequestUrl(this.__ldAutoLikeUrl)) {
                this.addEventListener("load", function () {
                    if (this.status !== 429) return;
                    try {
                        handleLikeRateLimit(JSON.parse(this.responseText));
                    } catch (_) {
                        handleLikeRateLimit(null);
                    }
                }, { once: true });
            }
            if (isReadTimingRequestUrl(this.__ldAutoLikeUrl)) {
                this.addEventListener("load", function () {
                    if (this.status < 200 || this.status >= 300) return;
                    const timing = extractReadTimingData(requestBody);
                    recordConfirmedReadTiming(timing.topicId, timing.postNumbers);
                }, { once: true });
            }
            return originalSend.apply(this, arguments);
        };
    }

    function parseCssPixels(value) {
        return Number.parseFloat(value || "0") || 0;
    }

    function isBlueTimelineColor(color) {
        const match = String(color || "").match(/rgba?\(([^)]+)\)/i);
        if (!match) return false;

        const parts = match[1].split(",").map(function (part) {
            return Number.parseFloat(part.trim());
        });
        const red = parts[0] || 0;
        const green = parts[1] || 0;
        const blue = parts[2] || 0;
        const alpha = parts[3] === undefined ? 1 : Number(parts[3]);
        return alpha > 0.35 && blue >= 120 && green >= 90 && red <= 80;
    }

    function isBlueDotStyle(style, fallbackRect) {
        if (!style || style.display === "none" || style.visibility === "hidden") return false;
        if (Number(style.opacity || 1) < 0.2) return false;

        const width = parseCssPixels(style.width) || (fallbackRect ? fallbackRect.width : 0);
        const height = parseCssPixels(style.height) || (fallbackRect ? fallbackRect.height : 0);
        const radius = parseCssPixels(style.borderTopLeftRadius);
        const isSmallRound = width >= 4 && width <= 18 && height >= 4 && height <= 18 &&
            Math.abs(width - height) <= 6 && radius >= Math.min(width, height) * 0.35;

        return isSmallRound && [
            style.backgroundColor,
            style.borderColor,
            style.color
        ].some(isBlueTimelineColor);
    }

    function hasUnreadTimelineBlueDot() {
        const containers = Array.from(document.querySelectorAll([
            ".topic-timeline",
            ".timeline-scrollarea",
            ".timeline-container",
            ".timeline-replies",
            ".topic-navigation"
        ].join(",")));

        return containers.some(function (container) {
            return Array.from(container.querySelectorAll("*")).some(function (element) {
                const rect = element.getBoundingClientRect();
                if (rect.width <= 0 || rect.height <= 0) return false;

                if (isBlueDotStyle(window.getComputedStyle(element), rect)) return true;

                return ["::before", "::after"].some(function (pseudo) {
                    const style = window.getComputedStyle(element, pseudo);
                    return style && style.content !== "none" && isBlueDotStyle(style, null);
                });
            });
        });
    }

    function getTopicProgressState() {
        const progress = document.querySelector("#topic-progress");
        if (!progress) return { available: false, current: 0, total: 0, complete: false };

        const values = Array.from(progress.querySelectorAll(".nums span"))
            .map(function (element) {
                return parseInt(element.textContent || "", 10);
            })
            .filter(Number.isFinite);

        if (values.length < 2) {
            const fallbackValues = String(progress.textContent || "").match(/\d+/g) || [];
            values.length = 0;
            fallbackValues.forEach(function (value) {
                const parsed = parseInt(value, 10);
                if (Number.isFinite(parsed)) values.push(parsed);
            });
        }

        const current = values[0] || 0;
        const total = values[1] || 0;
        return {
            available: total > 0,
            current: current,
            total: total,
            complete: total > 0 && current >= total
        };
    }

    function finishLikeRound(reason, messageText) {
        if (!running) return;
        savePendingTarget(null);
        saveFallbackGeneric(null);
        saveWhitelistProgress(null);
        if (stopRequested || isRunExpired()) {
            stopRun(stopReason || "已到设定时间，刷帖结束", true);
            return;
        }
        scheduleNextLike(Date.now());
        if (messageText) setMessage(messageText);
        goToLatest();
    }

    function navigateToCandidate(candidate, reason, sourceTopicId, triedUsers, searchDeadline) {
        if (!running || stopRequested || isRunExpired()) {
            if (running) requestGracefulStop("已到设定时间，刷帖结束");
            return;
        }
        savePendingTarget({
            candidate: candidate,
            reason: reason,
            sourceTopicId: sourceTopicId || null,
            triedUsers: normalizeTriedUsers(triedUsers),
            searchDeadline: normalizeLikeSearchDeadline(searchDeadline),
            whitelistProgress: whitelistProgress
        });

        stopReadTimer();
        const destination = candidate.url || buildCandidateUrl(candidate);
        const currentPath = location.pathname.replace(/\/$/, "");
        const destinationPath = new URL(destination, location.origin).pathname.replace(/\/$/, "");

        if (currentPath === destinationPath) {
            setTimeout(function () {
                processPendingTarget();
            }, 450);
        } else {
            location.href = destination;
        }
    }

    function goToLatest() {
        if (!running) return;
        if (shouldStopStartingNewWork()) {
            stopRun(stopReason || "已到设定时间，刷帖结束", true);
            return;
        }

        if (isLatestPage()) {
            setTimeout(function () {
                openNext();
            }, 500);
        } else {
            location.href = "/latest";
        }
    }

    function handleLikeApiError(error, reason) {
        if (!running) return;
        const label = error instanceof ForumLoginError ? "论坛没有登录" : "动态接口不可用";
        console.warn("[自动刷帖] 登录态或动态接口检查失败，本轮点赞跳过:", error);

        // 只清理当前点赞任务；不停止刷帖，也不关闭后续点赞。
        savePendingTarget(null);
        saveFallbackGeneric(null);
        saveWhitelistProgress(null);
        if (stopRequested || isRunExpired()) {
            stopRun(stopReason || "已到设定时间，刷帖结束", true);
            return;
        }
        scheduleNextLike(Date.now());
        setMessage(label + "，本次点赞跳过；刷帖继续，后续点赞会重新尝试");
        goToLatest();
    }

    function handleTopicApiError(error) {
        if (!running) return;
        console.warn("[自动刷帖] 热门帖子接口暂不可用:", error);
        setMessage("热门帖子接口暂不可用，刷帖继续，稍后重试");
        setTimeout(function () {
            if (!running) return;
            if (shouldStopStartingNewWork()) {
                stopRun(stopReason || "已到设定时间，刷帖结束", true);
                return;
            }
            if (isLatestPage()) {
                openNext();
            } else {
                goToLatest();
            }
        }, 5000);
    }

    async function processPendingTarget() {
        if (!running || !pendingTarget || targetActionRunning) return;
        if (shouldStopStartingNewWork()) {
            stopRun(stopReason || "已到设定时间，刷帖结束", true);
            return;
        }

        const task = pendingTarget;
        const candidate = task.candidate;
        const searchDeadline = normalizeLikeSearchDeadline(task.searchDeadline);
        if (!whitelistProgress && task.whitelistProgress) {
            saveWhitelistProgress(task.whitelistProgress);
        }
        const triedUsers = normalizeTriedUsers(task.triedUsers);
        if (candidate && candidate.username && !triedUsers.some(function (name) {
            return name.toLowerCase() === String(candidate.username).toLowerCase();
        })) {
            triedUsers.push(String(candidate.username));
        }

        if (isLikeSearchExpired(searchDeadline)) {
            finishLikeRound(
                task.reason,
                "本轮已寻找 1 分钟仍未确认点赞成功，先跳过点赞并继续刷帖"
            );
            return;
        }

        if (!isCandidateTopicPage(candidate)) {
            navigateToCandidate(
                candidate,
                task.reason,
                task.sourceTopicId,
                triedUsers,
                searchDeadline
            );
            return;
        }

        targetActionRunning = true;
        try {
            await ensureLoggedIn();
            if (!running) return;

            const result = await attemptTargetCandidate(candidate, searchDeadline);
            if (!running || result.status === "stopped") return;
            if (stopRequested || isRunExpired()) {
                finishLikeRound(task.reason);
                return;
            }

            if (result.status === "success") {
                finishLikeRound(task.reason);
                return;
            }

            if (result.status === "timeout" || isLikeSearchExpired(searchDeadline)) {
                finishLikeRound(
                    task.reason,
                    "本轮已寻找 1 分钟仍未确认点赞成功，先跳过点赞并继续刷帖"
                );
                return;
            }

            if (result.status === "rate_limited" || likeCooldownUntil > Date.now()) {
                finishLikeRound(task.reason, "点赞受到论坛限制，本轮先跳过并继续刷帖");
                return;
            }

            if (result.status === "self") {
                setMessage("当前登录账号在白名单中，跳过自己的内容并继续寻找其他用户");
            }

            const nextSelection = await chooseRandomWhitelistCandidate([], searchDeadline);
            if (!running) return;
            if (stopRequested || isRunExpired()) {
                finishLikeRound(task.reason);
                return;
            }

            if (nextSelection.timedOut || isLikeSearchExpired(searchDeadline)) {
                finishLikeRound(
                    task.reason,
                    "本轮已寻找 1 分钟仍未确认点赞成功，先跳过点赞并继续刷帖"
                );
                return;
            }

            if (nextSelection.candidate) {
                const sameUser = candidate && candidate.username &&
                    String(candidate.username).toLowerCase() ===
                    String(nextSelection.candidate.username).toLowerCase();
                setMessage(sameUser
                    ? "当前只有这一位白名单用户可用，继续随机其下一条内容"
                    : "当前内容已点赞、不可点赞或未确认，已随机切换到用户 " +
                        nextSelection.candidate.username);
                navigateToCandidate(
                    nextSelection.candidate,
                    task.reason,
                    task.sourceTopicId,
                    nextSelection.triedUsers,
                    searchDeadline
                );
                return;
            }

            savePendingTarget(null);
            saveFallbackGeneric({
                reason: task.reason,
                searchDeadline: searchDeadline
            });
            setMessage("名单中暂时没有可点赞内容，将转为普通随机点赞");
            goToLatest();
        } catch (error) {
            if (!running) return;
            if (error instanceof ForumLoginError || error instanceof ActivityApiError) {
                handleLikeApiError(error, task.reason);
            } else {
                console.warn("[自动刷帖] 指定用户点赞失败:", error);
                if (candidate && candidate.topicId) markLikeTopicHandled(candidate.topicId);
                try {
                    const nextSelection = await chooseRandomWhitelistCandidate([], searchDeadline);
                    if (stopRequested || isRunExpired()) {
                        finishLikeRound(task.reason);
                        return;
                    }
                    if (nextSelection.candidate && !isLikeSearchExpired(searchDeadline)) {
                        const sameUser = candidate && candidate.username &&
                            String(candidate.username).toLowerCase() ===
                            String(nextSelection.candidate.username).toLowerCase();
                        setMessage(sameUser
                            ? "当前只有这一位白名单用户可用，继续随机其下一条内容"
                            : "当前内容处理异常，已随机切换到用户 " +
                                nextSelection.candidate.username);
                        navigateToCandidate(
                            nextSelection.candidate,
                            task.reason,
                            task.sourceTopicId,
                            nextSelection.triedUsers,
                            searchDeadline
                        );
                        return;
                    }
                } catch (nextError) {
                    if (!running) return;
                    if (nextError instanceof ForumLoginError || nextError instanceof ActivityApiError) {
                        handleLikeApiError(nextError, task.reason);
                        return;
                    }
                    console.warn("[自动刷帖] 寻找下一条名单内容失败:", nextError);
                }
                savePendingTarget(null);
                saveFallbackGeneric({ reason: task.reason, searchDeadline: searchDeadline });
                setMessage("名单内容处理异常，转为普通随机点赞继续寻找");
                goToLatest();
            }
        } finally {
            targetActionRunning = false;
        }
    }

    async function processFallbackGeneric() {
        if (!running || !fallbackGeneric || targetActionRunning) return;
        if (shouldStopStartingNewWork()) {
            stopRun(stopReason || "已到设定时间，刷帖结束", true);
            return;
        }
        const task = fallbackGeneric;
        const searchDeadline = normalizeLikeSearchDeadline(task.searchDeadline);

        if (isLikeSearchExpired(searchDeadline)) {
            finishLikeRound(
                task.reason,
                "本轮已寻找 1 分钟仍未确认点赞成功，先跳过点赞并继续刷帖"
            );
            return;
        }
        if (!isTopicPage()) {
            goToLatest();
            return;
        }
        targetActionRunning = true;
        try {
            await ensureLoggedIn();
            if (!running) return;

            const topicId = getCurrentTopicId();
            const result = await attemptGenericLike(topicId, searchDeadline);
            if (!running || result.status === "stopped") return;
            if (stopRequested || isRunExpired()) {
                finishLikeRound(task.reason);
                return;
            }

            if (result.status === "success") {
                finishLikeRound(task.reason);
                return;
            }

            if (result.status === "timeout" || isLikeSearchExpired(searchDeadline)) {
                finishLikeRound(
                    task.reason,
                    "本轮已寻找 1 分钟仍未确认点赞成功，先跳过点赞并继续刷帖"
                );
                return;
            }

            if (result.status === "rate_limited" || likeCooldownUntil > Date.now()) {
                finishLikeRound(task.reason, "点赞受到论坛限制，本轮先跳过并继续刷帖");
                return;
            }

            saveFallbackGeneric({
                reason: task.reason,
                searchDeadline: searchDeadline
            });
            setMessage("本主题已点赞、不可点赞或未确认，继续寻找下一个帖子");
            goToLatest();
        } catch (error) {
            if (!running) return;
            if (error instanceof ForumLoginError || error instanceof ActivityApiError) {
                handleLikeApiError(error, task.reason);
            } else {
                console.warn("[自动刷帖] 普通随机点赞兜底失败:", error);
                saveFallbackGeneric({
                    reason: task.reason,
                    searchDeadline: searchDeadline
                });
                setMessage("当前帖子点赞发生异常，继续寻找下一个帖子");
                goToLatest();
            }
        } finally {
            targetActionRunning = false;
        }
    }

    function getRenderedTopicUrl(topicId) {
        const link = getRenderedTopicLink(topicId);
        if (!link) return null;

        try {
            const parsed = new URL(link.getAttribute("href"), location.origin);
            parsed.search = "";
            parsed.hash = "";
            return parsed.href;
        } catch (_) {
            return null;
        }
    }

    function getRenderedTopicLink(topicId) {

        return Array.from(document.querySelectorAll("a.title")).find(function (link) {
            const info = getTopicInfoFromUrl(link.href);
            return info && String(info.topicId) === id;
        }) || null;
    }

    async function trackTopicVisit(topicInfo) {
        const id = String(topicInfo && topicInfo.topicId || "").trim();
        if (!id) return false;

        try {
            // 与历史版本真实标题点击产生的请求保持一致。仅有 track_visit
            // 参数只会更新阅读位置，缺少以下追踪头不会计为浏览器主题访问。
            const postNumber = parseTopicNumber(topicInfo.postNumber);
            const postPath = postNumber > 1 ? "/" + postNumber : "";
            const csrfMeta = document.querySelector('meta[name="csrf-token"]');
            const headers = {
                "Accept": "application/json, text/javascript, */*; q=0.01",
                "X-Requested-With": "XMLHttpRequest",
                "Discourse-Logged-In": "true",
                "Discourse-Present": "true",
                "Discourse-Track-View": "true",
                "Discourse-Track-View-Topic-Id": id
            };
            if (csrfMeta && csrfMeta.content) {
                headers["X-CSRF-Token"] = csrfMeta.content;
            }

            const response = await fetch(
                "/t/" + encodeURIComponent(id) + postPath +
                ".json?track_visit=true&forceLoad=true",
                {
                    credentials: "same-origin",
                    cache: "no-store",
                    headers: headers
                }
            );

            // 读完整个响应，并要求论坛明确确认两类浏览追踪后再进入主题。
            await response.text();
            return response.ok &&
                response.headers.get("X-Discourse-TrackView") === "1" &&
                response.headers.get("X-Discourse-BrowserPageView") === "1";
        } catch (error) {
            console.warn("[自动刷帖] 主题浏览追踪请求失败:", error);
            return false;
        }
    }

    function isRenderedTopicFullyRead(topicId) {
        if (!isLatestPage() || !topicId) return false;

        const row = document.querySelector(
            'tr.topic-list-item[data-topic-id="' + String(topicId) + '"]'
        );
        if (!row) return false;

        const hasUnread = row.classList.contains("unread-posts") ||
            row.classList.contains("new-posts") ||
            !!row.querySelector(".badge-notification.new-posts, .badge-notification.unread-posts");
        return row.classList.contains("visited") && !hasUnread;
    }

    function parseTopicNumber(value) {
        const number = parseInt(value || "", 10);
        return Number.isFinite(number) && number > 0 ? number : 0;
    }

    function getTopicResumePostNumber(topic) {
        if (!topic) return 0;

        const lastRead = parseTopicNumber(topic.last_read_post_number);
        const highest = parseTopicNumber(
            topic.highest_post_number || topic.posts_count
        );
        const unread = Math.max(
            parseTopicNumber(topic.unread),
            parseTopicNumber(topic.unread_posts),
            parseTopicNumber(topic.new_posts)
        );

        // Discourse 的热门列表会把未读主题指向“最后已读楼层 + 1”。
        if (unread > 0 || (lastRead > 0 && highest > lastRead)) {
            return lastRead > 0 ? lastRead + 1 : 1;
        }

        // 已经读完的主题至少从最后阅读楼层进入，避免再次从首帖开始。
        if (lastRead > 0 && highest > 0 && lastRead >= highest) {
            return lastRead;
        }

        return 0;
    }

    function isTopicFullyRead(topic) {
        if (!topic || !topic.id) return false;
        if (isRenderedTopicFullyRead(topic.id)) return true;

        const lastRead = parseTopicNumber(topic.last_read_post_number);
        const highest = parseTopicNumber(
            topic.highest_post_number || topic.posts_count
        );
        const unread = Math.max(
            parseTopicNumber(topic.unread),
            parseTopicNumber(topic.unread_posts),
            parseTopicNumber(topic.new_posts)
        );

        return lastRead > 0 && highest > 0 && lastRead >= highest && unread === 0;
    }

    function appendTopicPostNumber(url, postNumber) {
        const number = parseTopicNumber(postNumber);
        if (!url || number <= 1) return url;

        try {
            const parsed = new URL(url, location.origin);
            const info = getTopicInfoFromUrl(parsed.href);
            if (!info || info.postNumber > 1) return parsed.href;
            parsed.pathname = parsed.pathname.replace(/\/$/, "") + "/" + number;
            parsed.search = "";
            parsed.hash = "";
            return parsed.href;
        } catch (_) {
            return url;
        }
    }

    function topicUrlFromJson(topic) {
        if (!topic || !topic.id) return null;

        // 页面已经渲染出的链接是 Discourse 最权威的“直达未读/上次阅读”结果，
        // 优先使用它，避免自行推算时遗漏站点的阅读状态规则。
        const renderedUrl = getRenderedTopicUrl(topic.id);
        if (renderedUrl) return renderedUrl;

        let baseUrl = null;
        if (topic.url) {
            try {
                const parsed = new URL(topic.url, location.origin);
                if (parsed.pathname.indexOf("/t/") === 0) {
                    parsed.search = "";
                    parsed.hash = "";
                    baseUrl = parsed.href;
                }
            } catch (_) {
                // 继续使用 slug + id 构造地址。
            }
        }

        if (!baseUrl) {
            const slug = String(topic.slug || "topic").trim() || "topic";
            baseUrl = location.origin + "/t/" + encodeURIComponent(slug) + "/" + String(topic.id);
        }

        return appendTopicPostNumber(baseUrl, getTopicResumePostNumber(topic));
    }

    async function fetchTopicList() {
        const topics = [];
        const seenIds = new Set();

        // /top.json 不附带 period 参数，使用论坛当前默认热门周期。
        for (let page = 0; page < CONFIG.MAX_TOP_PAGES; page++) {
            const result = await requestJson("/top.json?page=" + page + "&per_page=50");
            if (result.response.status === 401 || result.response.status === 403) {
                throw new ForumLoginError("论坛没有登录");
            }
            if (!result.response.ok || !result.data ||
                !result.data.topic_list || !Array.isArray(result.data.topic_list.topics)) {
                throw new ActivityApiError("/top 热门接口不可用");
            }

            const pageTopics = result.data.topic_list.topics;
            if (!pageTopics.length) break;

            pageTopics.forEach(function (topic) {
                const id = topic && topic.id ? String(topic.id) : "";
                if (!id || isTopicFullyRead(topic)) return;
                const url = topicUrlFromJson(topic);
                if (!url || seenIds.has(id)) return;
                seenIds.add(id);
                topics.push(url);
            });

            if (topics.length >= CONFIG.MAX_TOPICS) break;
        }

        // 保留以前进入过但尚未读完的热门主题；500 条候选统一随机抽取。
        topicList = shuffle(topics.slice(0, CONFIG.MAX_TOPICS));
        topicIndex = 0;
        writeJson(KEY.TOPIC_LIST, topicList);
        localStorage.setItem(KEY.TOPIC_INDEX, "0");
    }

    function getNextTopicUrl() {
        const available = [];
        topicList.forEach(function (url, index) {
            const info = getTopicInfoFromUrl(url);
            const sameTopicVisited = !!(
                info && visitedTopicIds.has(String(info.topicId))
            );
            if (url && !visitedTopics.has(url) && !sameTopicVisited) {
                available.push({ url: url, index: index });
            }
        });

        if (!available.length) return null;

        const selected = available[randomBetween(0, available.length - 1)];
        topicIndex = selected.index + 1;
        localStorage.setItem(KEY.TOPIC_INDEX, String(topicIndex));
        return selected.url;
    }

    function savePendingOpen(value) {
        pendingOpen = value || null;
        if (pendingOpen) {
            writeJson(KEY.PENDING_OPEN, pendingOpen);
        } else {
            localStorage.removeItem(KEY.PENDING_OPEN);
        }
    }

    function beginTopicOpen(primaryUrl, actualUrl) {
        const destination = actualUrl || primaryUrl;
        const info = getTopicInfoFromUrl(destination);
        if (!info) return false;

        savePendingOpen({
            topicId: String(info.topicId),
            primaryUrl: String(primaryUrl || destination),
            actualUrl: String(destination),
            countAsRead: !fallbackGeneric,
            startedAt: Date.now()
        });
        return true;
    }

    function confirmPendingOpen() {
        if (!pendingOpen) return false;
        const current = getCurrentTopicInfo();
        if (!current || String(current.topicId) !== String(pendingOpen.topicId)) return false;

        const task = pendingOpen;
        savePendingOpen(null);
        [task.primaryUrl, task.actualUrl].forEach(function (url) {
            if (!url) return;
            visitedTopics.add(String(url));
            const info = getTopicInfoFromUrl(url);
            if (info) visitedTopicIds.add(String(info.topicId));
        });
        writeJson(KEY.VISITED, Array.from(visitedTopics));

        if (task.countAsRead) {
            readCount++;
            localStorage.setItem(KEY.READ_COUNT, String(readCount));
            updateUI();
        }
        return true;
    }

    function armTopicOpenWatchdog(topicId) {
        const expectedId = String(topicId || "");
        setTimeout(function () {
            if (!running || !pendingOpen ||
                String(pendingOpen.topicId) !== expectedId) return;
            if (confirmPendingOpen()) return;

            savePendingOpen(null);
            setMessage("主题未成功打开，已重新选择");
            if (isLatestPage()) {
                openNext();
            } else {
                goToLatest();
            }
        }, 10000);
    }

    function getRandomRenderedTopicLink(excludedTopicId) {
        const excluded = String(excludedTopicId || "");
        const topIds = new Set(topicList.map(function (url) {
            const info = getTopicInfoFromUrl(url);
            return info ? String(info.topicId) : "";
        }).filter(Boolean));

        const candidates = Array.from(document.querySelectorAll("a.title"))
            .map(function (link) {
                return { link: link, info: getTopicInfoFromUrl(link.href) };
            })
            .filter(function (item) {
                return item.info && String(item.info.topicId) !== excluded;
            });
        const unvisited = candidates.filter(function (item) {
            return !visitedTopicIds.has(String(item.info.topicId));
        });
        const preferred = unvisited.filter(function (item) {
            return topIds.has(String(item.info.topicId));
        });
        const pool = preferred.length ? preferred : (unvisited.length ? unvisited : candidates);
        return pool.length ? pool[randomBetween(0, pool.length - 1)].link : null;
    }

    async function openNext() {
        if (!running || pendingTarget || pendingOpen || openNextInProgress) return;
        if (shouldStopStartingNewWork()) {
            stopRun(stopReason || "已到设定时间，刷帖结束", true);
            return;
        }
        if (!isLatestPage()) {
            location.href = "/latest";
            return;
        }

        openNextInProgress = true;
        try {
            if (!topicList.length) await fetchTopicList();
            if (!running || shouldStopStartingNewWork()) return;

            let desiredUrl = getNextTopicUrl();
            if (!desiredUrl) {
                // 热门列表可能在运行期间更新；重新请求一次后再决定是否循环阅读。
                await fetchTopicList();
                if (!running || shouldStopStartingNewWork()) return;
                desiredUrl = getNextTopicUrl();
            }

            if (!desiredUrl) {
                // 热门快照全部读过后允许循环阅读，但 handledLikeTopics 仍保证不重复点赞。
                if (topicList.length) {
                    const repeatIndex = randomBetween(0, topicList.length - 1);
                    desiredUrl = topicList[repeatIndex];
                    topicIndex = repeatIndex + 1;
                    localStorage.setItem(KEY.TOPIC_INDEX, String(topicIndex));
                }
            }

            if (!desiredUrl) {
                setMessage("/latest 暂无可读取主题，稍后重试");
                setTimeout(function () {
                    if (running) goToLatest();
                }, 1800);
                return;
            }

            if (shouldStopStartingNewWork()) return;

            const desiredInfo = getTopicInfoFromUrl(desiredUrl);
            const renderedLink = desiredInfo
                ? getRenderedTopicLink(desiredInfo.topicId)
                : null;

            if (renderedLink) {
                // 与最早版本一致，优先点击页面中的真实主题标题，
                // 让 Discourse 自己补齐浏览追踪请求头。
                if (!beginTopicOpen(desiredUrl, renderedLink.href)) return;
                realTopicClick(renderedLink);
                armTopicOpenWatchdog(desiredInfo.topicId);
                return;
            }

            if (desiredInfo && await trackTopicVisit(desiredInfo)) {
                if (!running || shouldStopStartingNewWork()) return;
                if (!beginTopicOpen(desiredUrl, desiredUrl)) return;
                armTopicOpenWatchdog(desiredInfo.topicId);
                location.href = desiredUrl;
                return;
            }

            if (!running || shouldStopStartingNewWork()) return;

            // 追踪未得到论坛双重确认时不直接跳转；沿用历史版本的做法，
            // 改点当前 /latest 页面中真实渲染出的随机标题。
            const fallbackLink = getRandomRenderedTopicLink(
                desiredInfo && desiredInfo.topicId
            );
            if (fallbackLink) {
                setMessage("主题浏览追踪未确认，已改用真实标题进入");
                const fallbackInfo = getTopicInfoFromUrl(fallbackLink.href);
                if (!fallbackInfo || !beginTopicOpen(fallbackLink.href, fallbackLink.href)) return;
                realTopicClick(fallbackLink);
                armTopicOpenWatchdog(fallbackInfo.topicId);
                return;
            }

            setMessage("主题列表尚未加载完成，稍后重试");
            setTimeout(function () {
                if (running) goToLatest();
            }, 1500);
        } catch (error) {
            if (!running) return;
            if (error instanceof ForumLoginError || error instanceof ActivityApiError) {
                handleTopicApiError(error);
                return;
            }
            console.warn("[自动刷帖] 打开下一帖子失败:", error);
            setMessage("获取帖子失败，稍后重试");
            setTimeout(function () {
                if (running) goToLatest();
            }, 1800);
        } finally {
            openNextInProgress = false;
        }
    }

    async function finishCurrentTopic(topicId) {
        const id = String(topicId);
        if (!running || finishingTopicId === id) return;
        finishingTopicId = id;

        try {
            if (shouldStopStartingNewWork()) {
                stopRun(stopReason || "已到设定时间，刷帖结束", true);
                return;
            }

            if (!isLikeDue()) {
                goToLatest();
                return;
            }

            const searchDeadline = createLikeSearchDeadline();

            try {
                await ensureLoggedIn();
                if (!running || shouldStopStartingNewWork()) return;
                beginWhitelistRound();
                const selection = await chooseRandomWhitelistCandidate([], searchDeadline);
                if (!running) return;
                if (stopRequested || isRunExpired()) {
                    stopRun(stopReason || "已到设定时间，刷帖结束", true);
                    return;
                }

                if (selection.timedOut || isLikeSearchExpired(searchDeadline)) {
                    finishLikeRound(
                        "scheduled",
                        "本轮已寻找 1 分钟仍未确认点赞成功，先跳过点赞并继续刷帖"
                    );
                    return;
                }

                if (selection.candidate) {
                    setMessage("本帖已刷完，准备优先点赞名单用户 " + selection.candidate.username);
                    navigateToCandidate(
                        selection.candidate,
                        "scheduled",
                        id,
                        selection.triedUsers,
                        searchDeadline
                    );
                    return;
                }
            } catch (error) {
                if (error instanceof ForumLoginError || error instanceof ActivityApiError) {
                    handleLikeApiError(error, "scheduled");
                    return;
                }
                throw error;
            }

            saveFallbackGeneric({
                reason: "scheduled",
                searchDeadline: searchDeadline
            });
            if (shouldStopStartingNewWork()) return;
            setMessage("名单中暂时没有可点赞内容，将转为普通随机点赞");
            await processFallbackGeneric();
        } catch (error) {
            if (!running) return;
            console.warn("[自动刷帖] 完成帖子流程失败:", error);
            finishLikeRound("scheduled", "本轮点赞流程发生异常，先跳过并继续刷帖");
        } finally {
            finishingTopicId = null;
        }
    }

    function getTopicPostNumber(post) {
        if (!post) return 0;

        const dataNodes = [post];
        const nestedData = post.querySelector("[data-post-number]");
        if (nestedData && nestedData !== post) dataNodes.push(nestedData);
        for (let i = 0; i < dataNodes.length; i++) {
            const value = parseTopicNumber(dataNodes[i].getAttribute("data-post-number"));
            if (value > 0) return value;
        }

        const idNodes = [post];
        const nestedId = post.querySelector('[id^="post_"]');
        if (nestedId && nestedId !== post) idNodes.push(nestedId);
        for (let i = 0; i < idNodes.length; i++) {
            const match = String(idNodes[i].id || "").match(/^post_(\d+)$/);
            if (match) {
                const value = parseTopicNumber(match[1]);
                if (value > 0) return value;
            }
        }

        return 0;
    }

    function getRenderedTopicPosts() {
        const posts = [];
        const seenNumbers = new Set();
        Array.from(document.querySelectorAll(".topic-post")).forEach(function (element) {
            if (!element || !element.isConnected) return;
            const number = getTopicPostNumber(element);
            if (!number || seenNumbers.has(number)) return;
            seenNumbers.add(number);
            posts.push({ element: element, number: number });
        });
        posts.sort(function (a, b) {
            return a.number - b.number;
        });
        return posts;
    }

    function isTopicPostPartlyVisible(post) {
        if (!post) return false;
        const rect = post.getBoundingClientRect();
        const root = getPageScrollRoot();
        const viewportHeight = root && root.clientHeight ? root.clientHeight : window.innerHeight;
        return rect.bottom > 0 && rect.top < viewportHeight;
    }

    function getReadPageMetrics() {
        const root = getPageScrollRoot();
        const height = root && Number.isFinite(root.scrollHeight) ? root.scrollHeight : 0;
        const viewportHeight = root && root.clientHeight ? root.clientHeight : window.innerHeight;
        const scrollTop = root && Number.isFinite(root.scrollTop) ? root.scrollTop : window.scrollY;
        return {
            height: height,
            viewportHeight: Math.max(1, viewportHeight || 1),
            scrollTop: scrollTop || 0,
            atBottom: (scrollTop || 0) + (viewportHeight || 1) >= height - 8
        };
    }

    function fastRead() {
        if (!running || pendingTarget || !isTopicPage()) return;
        const topicInfo = getCurrentTopicInfo();
        const topicId = topicInfo && topicInfo.topicId;
        if (!topicId) return;

        if (activeReadState && activeReadTopicId === topicId && !activeReadState.cancelled) return;
        stopReadTimer();

        const initialPostNumber = topicInfo.postNumber > 1 ? topicInfo.postNumber : 0;
        const state = {
            topicId: topicId,
            cancelled: false,
            phase: "opening",
            currentPostNumber: null,
            cursor: initialPostNumber > 1 ? initialPostNumber - 1 : 0,
            cursorInitialized: initialPostNumber > 1,
            processedNumbers: new Set(),
            readDeadline: Date.now() + CONFIG.MAX_TOPIC_READ_DURATION,
            bottomRounds: 0,
            lastBottomSignature: "",
            emptyRounds: 0,
            progressWaitRounds: 0,
            progressWaitSignature: ""
        };
        activeReadState = state;
        activeReadTopicId = topicId;

        function isActive() {
            return activeReadState === state && !state.cancelled && running &&
                !pendingTarget && getCurrentTopicId() === topicId;
        }

        function scheduleNext(delay) {
            if (!isActive()) return;
            activeReadTimer = setTimeout(function () {
                activeReadTimer = null;
                Promise.resolve(step()).catch(function (error) {
                    console.warn("[自动刷帖] 逐楼阅读状态异常:", error);
                    if (isActive()) {
                        state.currentPostNumber = null;
                        state.phase = "retry";
                        setMessage("页面正在重新布局，稍后继续阅读本帖");
                        scheduleNext(CONFIG.READ_CONFIRM_DELAY);
                    }
                });
            }, Math.max(0, delay || 0));
        }

        function clearActiveRead() {
            state.cancelled = true;
            if (activeReadTimer) {
                clearTimeout(activeReadTimer);
                activeReadTimer = null;
            }
            if (activeReadState === state) activeReadState = null;
            if (activeReadTopicId === topicId) activeReadTopicId = null;
        }

        function finishRead(reason) {
            if (activeReadState !== state) return;
            clearActiveRead();

            if (isRunExpired() && !stopRequested) {
                stopRequested = true;
                stopReason = "已到设定时间，刷帖结束";
                saveStopRequest();
            }

            if (stopRequested) {
                stopRun(stopReason || "已到设定时间，刷帖结束", true);
                return;
            }

            setMessage(reason || "本帖已完成逐楼阅读，继续下一个帖子");
            finishCurrentTopic(topicId);
        }

        function findPost(number) {
            return getRenderedTopicPosts().find(function (item) {
                return item.number === number;
            });
        }

        async function waitForScrollStable(number) {
            const startedAt = Date.now();
            let lastTop = null;
            let lastScrollTop = null;
            let stableRounds = 0;
            let sawTarget = false;
            let sawVisibleTarget = false;

            while (isActive() && Date.now() - startedAt < 1000) {
                await sleep(CONFIG.READ_SCROLL_SETTLE_DELAY);
                if (!isActive()) return false;

                const current = findPost(number);
                if (!current) continue;
                sawTarget = true;
                const rect = current.element.getBoundingClientRect();
                const metrics = getReadPageMetrics();
                const top = Number.isFinite(rect.top) ? rect.top : 0;
                const scrollTop = metrics.scrollTop;
                const visible = isTopicPostPartlyVisible(current.element);
                if (visible) sawVisibleTarget = true;

                if (lastTop !== null && Math.abs(top - lastTop) <= 2 &&
                    Math.abs(scrollTop - lastScrollTop) <= 2 && visible) {
                    stableRounds++;
                } else {
                    stableRounds = 0;
                }
                lastTop = top;
                lastScrollTop = scrollTop;

                if (stableRounds >= 2) return true;
            }

            return isActive() && sawTarget && sawVisibleTarget && stableRounds >= 1;
        }

        async function dwellVisible(milliseconds) {
            let remaining = milliseconds;
            let previous = Date.now();
            while (remaining > 0 && isActive()) {
                await sleep(Math.min(100, remaining));
                const now = Date.now();
                const visible = typeof document.visibilityState === "undefined" ||
                    document.visibilityState === "visible";
                if (visible) remaining -= now - previous;
                previous = now;
            }
            return isActive();
        }

        async function step() {
            if (!isActive()) return;

            if (!stopRequested && isRunExpired()) {
                requestGracefulStop("已到设定时间，刷帖结束");
                if (!isActive()) return;
            }

            if (stopRequested && state.currentPostNumber === null) {
                finishRead(stopReason || "已到设定时间，刷帖结束");
                return;
            }

            if (Date.now() >= state.readDeadline && state.currentPostNumber === null) {
                finishRead("本帖已阅读 3 分钟，继续下一个帖子");
                return;
            }

            let posts = getRenderedTopicPosts();
            if (!state.cursorInitialized) {
                const visiblePosts = posts.filter(function (item) {
                    return isTopicPostPartlyVisible(item.element);
                });
                if (visiblePosts.length) {
                    state.cursor = visiblePosts[0].number - 1;
                }
                state.cursorInitialized = true;
            }

            const next = posts.filter(function (item) {
                return item.number > state.cursor && !state.processedNumbers.has(item.number);
            })[0];

            if (next) {
                state.emptyRounds = 0;
                state.progressWaitRounds = 0;
                state.progressWaitSignature = "";
                state.bottomRounds = 0;
                state.lastBottomSignature = "";
                state.currentPostNumber = next.number;
                state.phase = "settling";

                const target = findPost(next.number);
                if (!target || !target.element) {
                    state.currentPostNumber = null;
                    scheduleNext(CONFIG.READ_SCROLL_SETTLE_DELAY);
                    return;
                }

                try {
                    target.element.scrollIntoView({
                        behavior: "auto",
                        block: "center",
                        inline: "nearest"
                    });
                } catch (_) {
                    target.element.scrollIntoView();
                }

                const stable = await waitForScrollStable(next.number);
                if (!isActive()) return;
                if (!stable) {
                    state.currentPostNumber = null;
                    state.phase = "retry";
                    scheduleNext(CONFIG.READ_SCROLL_SETTLE_DELAY);
                    return;
                }

                state.phase = "dwelling";
                const dwellTime = randomBetween(
                    CONFIG.READ_POST_DWELL_MIN,
                    CONFIG.READ_POST_DWELL_MAX
                );
                const dwellCompleted = await dwellVisible(dwellTime);
                if (!dwellCompleted || !isActive()) return;

                state.processedNumbers.add(next.number);
                state.cursor = Math.max(state.cursor, next.number);
                state.phase = "flushing";

                // 楼层停留已经完成；短暂等待本轮原生 timings 请求发出/返回。
                await sleep(CONFIG.READ_TIMING_FLUSH_DELAY);
                if (!isActive()) return;
                state.currentPostNumber = null;
                state.phase = "idle";

                if (stopRequested || isRunExpired()) {
                    if (!stopRequested) {
                        stopRequested = true;
                        stopReason = "已到设定时间，刷帖结束";
                        saveStopRequest();
                    }
                    finishRead(stopReason || "已到设定时间，刷帖结束");
                    return;
                }

                if (Date.now() >= state.readDeadline) {
                    finishRead("本帖已阅读 3 分钟，继续下一个帖子");
                    return;
                }

                scheduleNext(0);
                return;
            }

            // 没有新的已渲染楼层时，只做视口比例的小幅加载推进；推进本身不计为已读。
            state.currentPostNumber = null;
            state.phase = "loading";
            if (!posts.length) {
                state.emptyRounds++;
                scheduleNext(Math.min(500, CONFIG.READ_CONFIRM_DELAY));
                return;
            }

            const metrics = getReadPageMetrics();
            const lastPost = posts[posts.length - 1];
            const progress = getTopicProgressState();
            const lastProcessed = !!lastPost && state.processedNumbers.has(lastPost.number);
            const lastPostVisible = !!lastPost && isTopicPostPartlyVisible(lastPost.element);
            const signature = [
                lastPost ? lastPost.number : 0,
                posts.length,
                metrics.height
            ].join(":");

            // 超小窗口可能永远到不了 document 的真正底部（末帖后还有页脚）。
            // 只要最后一楼已完整停留、仍与视口相交且论坛进度已完成，就可稳定确认结束。
            const progressCompleteVisible = lastProcessed && progress.available &&
                progress.complete && lastPostVisible;
            if (progressCompleteVisible) {
                if (signature === state.lastBottomSignature) {
                    state.bottomRounds++;
                } else {
                    state.lastBottomSignature = signature;
                    state.bottomRounds = 1;
                }

                if (state.bottomRounds >= CONFIG.BOTTOM_SETTLE_CHECKS) {
                    finishRead("本帖已到最后楼层并确认阅读进度，继续下一个帖子");
                    return;
                }

                nudgeScrollForReadConfirmation();
                scheduleNext(CONFIG.READ_CONFIRM_DELAY);
                return;
            }

            // 末楼刚完成时，先给论坛进度一到两次 tick 的更新时间，避免立刻把末楼推离视口。
            if (lastProcessed && lastPostVisible && progress.available && !progress.complete) {
                if (state.progressWaitSignature !== signature) {
                    state.progressWaitSignature = signature;
                    state.progressWaitRounds = 0;
                }
                if (state.progressWaitRounds < 2) {
                    state.progressWaitRounds++;
                    nudgeScrollForReadConfirmation();
                    scheduleNext(CONFIG.READ_CONFIRM_DELAY);
                    return;
                }
            }

            if (!metrics.atBottom) {
                state.bottomRounds = 0;
                state.lastBottomSignature = "";
                const nudge = Math.max(40, Math.floor(metrics.viewportHeight * CONFIG.READ_LOAD_NUDGE_RATIO));
                scrollReadingPageBy(nudge, "auto");
                scheduleNext(CONFIG.READ_SCROLL_SETTLE_DELAY + 120);
                return;
            }

            if (signature === state.lastBottomSignature) {
                state.bottomRounds++;
            } else {
                state.lastBottomSignature = signature;
                state.bottomRounds = 1;
            }

            const unreadDot = hasUnreadTimelineBlueDot();
            const completeEvidence = lastProcessed && (
                (progress.available && progress.complete) ||
                (!progress.available && !unreadDot)
            );

            if (!completeEvidence) state.bottomRounds = 0;

            if (state.bottomRounds >= CONFIG.BOTTOM_SETTLE_CHECKS) {
                finishRead("本帖已到最后楼层并确认阅读进度，继续下一个帖子");
                return;
            }

            // 极小窗口可能把页脚留在最后楼层之后，轻微回滚/前进以触发最后一次计时或加载。
            nudgeScrollForReadConfirmation();
            scheduleNext(CONFIG.READ_CONFIRM_DELAY);
        }

        // 先等待主题打开约 600ms，再定位第一楼。
        scheduleNext(CONFIG.TOPIC_OPEN_DELAY);
    }

    function resetForNewRun(minutes) {
        clearPersistedTaskState(false);
        stopReadTimer();

        running = true;
        runMinutes = Number.isFinite(minutes) ? Math.max(0, minutes) : 0;
        startTime = 0;
        endTime = 0;
        readCount = 0;
        readReplyCount = 0;
        confirmedReadReplies = new Set();
        topicIndex = 0;
        topicList = [];
        visitedTopics = new Set();
        visitedTopicIds = new Set();

        likeCount = 0;
        nextLikeAt = 0;
        likeCooldownUntil = 0;
        handledLikeTopics = new Set();
        whitelistNames = [];
        whitelistRefreshAt = 0;
        pendingTarget = null;
        pendingOpen = null;
        fallbackGeneric = null;
        whitelistProgress = null;
        lastWhitelistUser = "";
        loggedInUsername = "";

        finishingTopicId = null;
        targetActionRunning = false;
        likeClickInProgress = false;
        stopRequested = false;
        stopReason = "已到设定时间，刷帖结束";
        startBrushingTimer();
        persistRunState();
        updateUI();
    }

    async function startRun(minutes) {
        if (startInProgress) return;
        startInProgress = true;
        resetForNewRun(minutes);
        setMessage("正在获取 GitHub 名单并检查论坛登录状态（本轮最多寻找 1 分钟）...");
        const searchDeadline = createLikeSearchDeadline();

        try {
            await refreshWhitelistNames(true);
            if (!running || shouldStopStartingNewWork()) return;
            await ensureLoggedIn();
            if (!running || shouldStopStartingNewWork()) return;

            beginWhitelistRound();
            const selection = await chooseRandomWhitelistCandidate([], searchDeadline);
            if (!running) return;
            if (stopRequested || isRunExpired()) {
                stopRun(stopReason || "已到设定时间，刷帖结束", true);
                return;
            }

            if (selection.timedOut || isLikeSearchExpired(searchDeadline)) {
                finishLikeRound(
                    "startup",
                    "启动点赞已寻找 1 分钟仍未确认成功，本次先跳过并开始刷帖"
                );
                return;
            }

            if (selection.candidate) {
                setMessage("启动刷帖前，优先点赞名单用户 " + selection.candidate.username);
                navigateToCandidate(
                    selection.candidate,
                    "startup",
                    null,
                    selection.triedUsers,
                    searchDeadline
                );
                return;
            }

            saveFallbackGeneric({
                reason: "startup",
                searchDeadline: searchDeadline
            });
            setMessage("名单中暂时没有可点赞内容，将先进行普通随机点赞");
            goToLatest();
        } catch (error) {
            if (!running) return;
            if (error instanceof ForumLoginError || error instanceof ActivityApiError) {
                handleLikeApiError(error, "startup");
            } else {
                console.warn("[自动刷帖] 启动失败:", error);
                stopRun("启动失败，已结束刷帖");
            }
        } finally {
            startInProgress = false;
        }
    }

    function stopRun(reason, force) {
        // 防止 requestGracefulStop 已同步结束后，调用方再次覆盖最终统计。
        if (!running) return;

        const readFloorInProgress = !!(
            activeReadState &&
            activeReadState.currentPostNumber !== null &&
            activeReadState.currentPostNumber !== undefined
        );
        if (!force && (readFloorInProgress || likeClickInProgress)) {
            requestGracefulStop(reason);
            return;
        }

        const totalRun = startTime ? fmtCompact(Date.now() - startTime) : "0:00";
        const totalRead = readCount;
        const totalReplies = readReplyCount;
        const totalLikes = likeCount;

        running = false;
        startInProgress = false;
        openNextInProgress = false;
        targetActionRunning = false;
        likeClickInProgress = false;
        finishingTopicId = null;
        stopReadTimer();

        startTime = 0;
        endTime = 0;
        runMinutes = 0;
        topicIndex = 0;
        topicList = [];
        visitedTopics = new Set();
        visitedTopicIds = new Set();
        nextLikeAt = 0;
        likeCooldownUntil = 0;
        handledLikeTopics = new Set();
        whitelistNames = [];
        whitelistRefreshAt = 0;
        pendingTarget = null;
        pendingOpen = null;
        fallbackGeneric = null;
        whitelistProgress = null;
        lastWhitelistUser = "";
        loggedInUsername = "";
        stopRequested = false;
        stopReason = "";

        localStorage.setItem(KEY.RUN, "0");
        localStorage.setItem(KEY.START, "0");
        localStorage.setItem(KEY.END, "0");
        localStorage.setItem(KEY.RUN_MINUTES, "0");
        localStorage.setItem(KEY.READ_COUNT, String(totalRead));
        localStorage.setItem(KEY.READ_REPLY_COUNT, String(totalReplies));
        writeJson(KEY.READ_REPLY_KEYS, Array.from(confirmedReadReplies));
        localStorage.setItem(KEY.LIKE_COUNT, String(totalLikes));
        localStorage.removeItem(KEY.TOPIC_INDEX);
        localStorage.removeItem(KEY.TOPIC_LIST);
        localStorage.removeItem(KEY.VISITED);
        localStorage.removeItem(KEY.NEXT_LIKE);
        localStorage.removeItem(KEY.LIKE_COOLDOWN);
        localStorage.removeItem(KEY.HANDLED_TOPICS);
        localStorage.removeItem(KEY.WHITELIST_NAMES);
        localStorage.removeItem(KEY.WHITELIST_REFRESH);
        localStorage.removeItem(KEY.PENDING_TARGET);
        localStorage.removeItem(KEY.PENDING_OPEN);
        localStorage.removeItem(KEY.FALLBACK_GENERIC);
        localStorage.removeItem(KEY.WHITELIST_PROGRESS);
        localStorage.removeItem(KEY.LAST_WHITELIST_USER);
        localStorage.removeItem(KEY.STOP_REQUESTED);
        localStorage.removeItem(KEY.STOP_REASON);

        setMessage(
            "赞" + totalLikes + " 题" + totalRead + " 回" + totalReplies +
            " 时" + totalRun + "｜" + (reason || "已停止刷帖")
        );
        updateUI();
    }

    async function handleCurrentPage() {
        if (!running) return;

        // 先确认刚刚是否已真正进入目标主题，再处理到时停止，避免面板漏记。
        if (pendingOpen && isTopicPage()) confirmPendingOpen();

        if (stopRequested || isRunExpired()) {
            requestGracefulStop(stopReason || "已到设定时间，刷帖结束");
            return;
        }

        if (pendingOpen) {
            if (!confirmPendingOpen()) {
                // 页面重载后仍停在列表页，说明上一次打开没有成功；清理后重试。
                if (isLatestPage()) {
                    savePendingOpen(null);
                    setMessage("上一次主题未成功打开，已重新选择");
                } else if (isTopicPage()) {
                    savePendingOpen(null);
                    setMessage("打开的主题与目标不一致，已返回列表重试");
                    goToLatest();
                    return;
                }
            }
        }

        if (pendingTarget) {
            if (isCandidateTopicPage(pendingTarget.candidate)) {
                await processPendingTarget();
            } else {
                navigateToCandidate(
                    pendingTarget.candidate,
                    pendingTarget.reason,
                    pendingTarget.sourceTopicId,
                    pendingTarget.triedUsers,
                    pendingTarget.searchDeadline
                );
            }
            return;
        }

        if (fallbackGeneric) {
            await processFallbackGeneric();
            return;
        }

        ensureLikeSchedule();

        if (isLatestPage()) {
            await openNext();
        } else if (isTopicPage()) {
            fastRead();
        } else {
            goToLatest();
        }
    }

    installLikeLimitInterceptor();
    createUI();

    setInterval(updateUI, 1000);

    setInterval(function () {
        if (running && isRunExpired()) {
            requestGracefulStop("已到设定时间，刷帖结束");
        }
    }, 1000);

    let lastRouteKey = getRouteKey();
    setInterval(function () {
        const routeKey = getRouteKey();
        if (routeKey === lastRouteKey) return;
        lastRouteKey = routeKey;
        stopReadTimer();
        setTimeout(function () {
            handleCurrentPage();
        }, 50);
    }, 300);

    if (running) {
        setTimeout(function () {
            handleCurrentPage();
        }, 50);
    }
})();
